# ---------------------------------------------------------
# WEB FRAMEWORK
# Flask: Lightweight Python web framework.
# We chose Flask over Django because:
#   - Django is a "batteries-included" framework designed for
#     large CRUD apps. It enforces its own ORM, admin, auth.
#   - Flask is minimal. We control every component ourselves.
#   - For a data science project where the logic is in Python
#     pipelines (not web forms), Flask is a much cleaner fit.
# python-dotenv: Loads .env file into environment variables
#   so Flask can read MONGO_URI, SECRET_KEY, etc. without
#   those values being hardcoded anywhere in the source code.
# ---------------------------------------------------------
Flask==3.0.3
python-dotenv==1.0.1


# ---------------------------------------------------------
# DATABASE
# pymongo: Official MongoDB driver for Python.
#   Lets Flask connect to MongoDB Atlas and perform
#   CRUD operations (Create, Read, Update, Delete).
# dnspython: Required when connecting to MongoDB Atlas
#   using the modern "mongodb+srv://" connection string.
#   Without it, PyMongo cannot resolve Atlas DNS hostnames.
# ---------------------------------------------------------
pymongo==4.7.3
dnspython==2.6.1


# ---------------------------------------------------------
# DATA MANIPULATION
# pandas: The core data manipulation library in Python.
#   Provides DataFrame — a table-like structure perfect
#   for loading CSVs, inspecting columns, cleaning data,
#   and feature engineering. Used throughout every phase.
# numpy: Numerical computing foundation.
#   Pandas is built on NumPy. We use it directly for
#   array operations, mathematical computations, and
#   handling numerical data in the ML pipeline.
# openpyxl: Needed if the Kaggle dataset is in .xlsx format.
#   Pandas uses this under the hood to read Excel files.
# ---------------------------------------------------------
pandas==2.2.2
numpy==1.26.4
openpyxl==3.1.4


# ---------------------------------------------------------
# DATA VISUALIZATION
# matplotlib: The foundational Python plotting library.
#   Produces static charts (bar, line, scatter, histogram).
#   Used directly and also as the backend for seaborn.
# seaborn: Statistical data visualization built on matplotlib.
#   Makes it much easier to produce attractive, informative
#   statistical graphics (heatmaps, count plots, box plots)
#   with far less code than raw matplotlib.
# ---------------------------------------------------------
matplotlib==3.9.0
seaborn==0.13.2


# ---------------------------------------------------------
# MACHINE LEARNING & NLP
# scikit-learn: The standard Python ML library.
#   We use it specifically for:
#     - TfidfVectorizer: Converts text into TF-IDF vectors
#     - cosine_similarity: Measures similarity between vectors
#     - train_test_split, metrics: For evaluation
#   We are NOT using it to train a classifier — we're using
#   its NLP and similarity utilities for recommendation.
# nltk: Natural Language Toolkit.
#   Provides: tokenization, stop words, lemmatization (WordNet).
#   We use it in the NLP preprocessing pipeline to clean
#   text before feeding it into TF-IDF.
# ---------------------------------------------------------
scikit-learn==1.5.0
nltk==3.8.1


# ---------------------------------------------------------
# UTILITIES
# tqdm: Progress bars for loops.
#   When processing 3400 schemes or building TF-IDF matrices,
#   a progress bar lets you know the pipeline is alive.
# Jinja2: HTML templating engine.
#   Flask uses Jinja2 to render .html templates with dynamic
#   Python data. It comes with Flask but we pin it explicitly
#   for reproducibility.
# Werkzeug: Flask's underlying WSGI utility library.
#   Handles routing, request/response objects, security
#   helpers. Also comes with Flask — pinned for stability.
# ---------------------------------------------------------
tqdm==4.66.4
Jinja2==3.1.4
Werkzeug==3.0.3
